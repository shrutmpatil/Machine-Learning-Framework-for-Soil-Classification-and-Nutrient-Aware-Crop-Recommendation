def correct_image_orientation(image):
    try:
        exif = image._getexif()
        if exif:
            for orientation in ExifTags.TAGS.keys():
                if ExifTags.TAGS[orientation] == "Orientation":
                    break
            orientation_value = exif.get(orientation, None)
            if orientation_value == 3:
                image = image.rotate(180, expand=True)
            elif orientation_value == 6:
                image = image.rotate(270, expand=True)
            elif orientation_value == 8:
                image = image.rotate(90, expand=True)
    except Exception as e:
        print("Orientation correction error:", e)
    return image

@app.route("/register", methods=["GET", "POST"])
def register():
    if request.method == "POST":
        username = request.form["username"]
        email = request.form["email"]
        password = generate_password_hash(request.form["password"])
        user = User(username=username, email=email, password=password)
        db.session.add(user)
        db.session.commit()
        flash("Account created! Please log in.", "success")
        return redirect(url_for("login"))
    return render_template("register.html")


@app.route("/login", methods=["GET", "POST"])
def login():
    if request.method == "POST":
        user = User.query.filter_by(email=request.form["email"]).first()
        if user and check_password_hash(user.password, request.form["password"]):
            login_user(user)
            return redirect(url_for("dashboard"))
        else:
            flash("Invalid credentials", "danger")
    return render_template("login.html")

# -------- Prediction --------
@app.route("/predict", methods=["GET", "POST"])
@login_required
def predict():
    prediction_result = None
    image_preview = None

    if request.method == "POST":
        try:
            print("Starting prediction process...")
            file = request.files.get("image")
            if not file or file.filename == "":
                flash("Please upload an image.", "danger")
                return render_template("predict.html")

            filename = secure_filename(file.filename)
            filepath = os.path.join(app.config["UPLOAD_FOLDER"], filename)
            file.save(filepath)
            print(f"Image saved to: {filepath}")

            img = Image.open(filepath).convert("RGB")
            img = correct_image_orientation(img)
            img_resized = img.resize((224, 224))
            img_array = np.expand_dims(np.array(img_resized) / 255.0, axis=0)
            print("Image processed.")

            # Soil prediction
            soil_pred = soil_model.predict(img_array)
            soil_label_str = soil_encoder.inverse_transform([np.argmax(soil_pred)])[0]
            print(f"Soil type predicted: {soil_label_str}")

            image_preview = os.path.join("uploads", filename).replace(os.sep, "/")

            # NPK + pH
            try:
                N = float(request.form.get("nitrogen", 0))
                P = float(request.form.get("phosphorus", 0))
                K = float(request.form.get("potassium", 0))
                pH = float(request.form.get("ph", 7))
                print(f"NPK pH: {N}, {P}, {K}, {pH}")
            except ValueError:
                flash("Invalid numeric values entered!", "danger")
                return render_template("predict.html")

            nutrients = np.array([[N, P, K, pH]])
            nutrients_scaled = scaler.transform(nutrients)
            soil_encoded = soil_encoder.transform([soil_label_str])[0]
            features = np.hstack((nutrients_scaled, [[soil_encoded]]))
            print("Features prepared.")

            # Crop prediction
            crop_pred = crop_model.predict(features)
            crop_names = crop_encoder.inverse_transform(crop_pred)[0]
            crop_name = ", ".join(crop_names)
            print(f"Crop predicted: {crop_name}")

            prediction_result = {
                "soil_type": soil_label_str,
                "recommended_crop": crop_name,
            }

            # Save history
            history = PredictionHistory(
                nitrogen=N,
                phosphorus=P,
                potassium=K,
                ph=pH,
                soil_type=soil_label_str,
                crop=crop_name,
                image_path=filepath,
                author=current_user,
            )
            db.session.add(history)
            db.session.commit()
            print("History saved.")

        except Exception as e:
            db.session.rollback()
            print(f"Prediction error: {str(e)}")
            flash(f"Error: {str(e)}", "danger")

    return render_template(
        "predict.html", prediction=prediction_result, image_preview=image_preview
    )

# -------- Download History as Excel --------
@app.route("/download_history")
@login_required
def download_history():
    records = PredictionHistory.query.filter_by(user_id=current_user.id).all()
    data = [
        {
            "Date": r.date,
            "Nitrogen": r.nitrogen,
            "Phosphorus": r.phosphorus,
            "Potassium": r.potassium,
            "pH": r.ph,
            "Soil Type": r.soil_type,
            "Crop": r.crop,
            "Image Filename": os.path.basename(r.image_path) if r.image_path else "",
        }
        for r in records
    ]
    df = pd.DataFrame(data)
    output = io.BytesIO()
    with pd.ExcelWriter(output) as writer:
        df.to_excel(writer, index=False, sheet_name="History")
    output.seek(0)
    return send_file(
        output,
        download_name="prediction_history.xlsx",
        as_attachment=True,
        mimetype="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
    )
